%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Page layout
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%
% The class sets DIV = 10, which gives the text only 7/10 of the page
% width. Raising DIV widens the text block and narrows the margins.
% Higher value = wider text. Keep an eye on the line length: beyond
% ~80 characters per line readability starts to suffer.
%
%   DIV = 12  ->  150mm text  (recommended)
%   DIV = 14  ->  157mm text  (noticeably wider)
%
% BCOR is the binding correction (extra inner margin for the spine).
% Set it to 0mm if the thesis will only ever be read as a PDF.

\KOMAoptions{DIV = 12}
