\chapter{Additional pre-fit distributions}
\label{app:prefit}
%
This appendix collects the pre-fit distributions that complement those shown in Section \ref{sec:bbtt_controlplots}. They are obtained with the same selection, corrections, and uncertainty band as in that section, for \textcolor{red}{[2024 / the complete 2022--2024 dataset]}, and are grouped into the control regions, where the corrections of the two dominant backgrounds are tested, and the categories of the analysis, where the complete prediction is compared with the data before the fit.

\section{Control regions}
\label{app:prefit_cr}
%
\textcolor{red}{[Additional observables in the Drell--Yan-enriched region, in the $ee$ and $\mu\mu$ channels, and in the $t\bar{t}$-enriched region of Section \ref{subsec:bbtt_cr}: $p_T$ and $\eta$ of the two leptons and of the two jets of the $H\to b\bar{b}$ candidate, $b$ tagging scores, numbers of jets and of $b$-tagged jets, $p_T^\text{miss}$, $\Delta R$ between the leptons, $p_T$ of the $HH$ system, and VBF jet pair variables.]}
